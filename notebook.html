# Encode categorical features
categorical_cols = df_cleaned.select_dtypes(include=['object']).columns
label_encoders = {}
for col in categorical_cols:
    le = LabelEncoder()
    df_cleaned[col] = le.fit_transform(df_cleaned[col])
    label_encoders[col] = le  # Save for decoding later

# Normalize numerical features (excluding churn label)
features = df_cleaned.drop(columns=['Attrition_Flag'])
scaler = StandardScaler()
features_scaled = scaler.fit_transform(features)
features_scaled_df = pd.DataFrame(features_scaled, columns=features.columns)
In [4]
# Elbow method: compute inertia for K = 1-10
inertia_values = {}
for k in range(1, 11):
    kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)
    kmeans.fit(features_scaled)
    inertia_values[k] = kmeans.inertia_

# Print the inertia values
print("Inertia values:")
for k, inertia in inertia_values.items():
    print(f"K = {k} - Inertia = {inertia:.2f}")

# Plot the elbow graph with values labeled
plt.figure(figsize=(10, 6))
plt.plot(list(inertia_values.keys()), list(inertia_values.values()), marker='o')
plt.title("Elbow Method for Optimal K (Inertia Labeled)")
plt.xlabel("Number of Clusters (K)")
plt.ylabel("Inertia (WCSS)")
plt.xticks(range(1, 11))

# Label each point with its inertia value
for k, inertia in inertia_values.items():
    plt.text(k, inertia + 1000, f"{int(inertia)}", ha='center', fontsize=9)

plt.grid(True)
plt.tight_layout()
plt.show()
